# Step 1: generate average rent per square meter by city and year
# Group the apartment data by city and year
rents_line_graph <- rents %>%
  group_by(city, year) %>%

  # Step 2: calculate the number of observations and average rent per square meter
  summarise(
    count = n(),
    price_sqm = mean(rent / size, na.rm = TRUE)
  ) %>%

  # Step 3: keep city-year combinations with enough observations
  # Focus on Berlin, Hamburg, and Munich
  filter(
    count > 20,
    city %in% c("Berlin", "Hamburg", "Munich")
  ) %>%

  # Step 4: keep only the variables needed for the graph
  select(year, city, price_sqm)

## Exercise 2: Construction Year and Lift Availability
# Step 1: create price per square meter for each apartment
# Divide monthly rent by apartment size
rents_ex2 <- rents %>%
  mutate(price_sqm = rent / size)


# Step 2: remove observations with missing values
# We need valid values for construction year, price per square meter, and lift
rents_ex2 <- rents_ex2 %>%
  filter(
    !is.na(construction),
    !is.na(price_sqm),
    !is.na(lift)
  )


# Step 3: focus on apartments built from 1850 onwards
# Very old buildings have very few observations and make the graph harder to interpret
rents_ex2 <- rents_ex2 %>%
  filter(construction >= 1850)


# Step 4: group construction years into 10-year periods
# This reduces noise and makes the relationship easier to see
rents_ex2 <- rents_ex2 %>%
  mutate(construction_period = floor(construction / 10) * 10)


# Step 5: calculate average rent per square meter
# Calculate the average separately for each construction period and lift category
rents_ex2_summary <- rents_ex2 %>%
  group_by(construction_period, lift) %>%
  summarise(
    avg_price_sqm = mean(price_sqm, na.rm = TRUE),
    count = n(),
    .groups = "drop"
  )


# Step 6: keep groups with enough observations
# This avoids showing averages based on only a very small number of apartments
rents_ex2_summary <- rents_ex2_summary %>%
  filter(count > 20)
